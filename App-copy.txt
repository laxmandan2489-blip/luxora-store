import { useEffect, useMemo, useState } from "react";
import "./App.css";

const API = "http://127.0.0.1:5000";

const categories = [
  "All Bags",
  "Bags",
  "Handbags",
  "Sling Bags",
  "Tote Bags",
  "Backpacks",
  "Laptop Bags",
  "Travel Bags",
  "Clutches",
  "Wallets",
  "Girls",
  "Boys",
  "Jewellery",
  "Rings",
  "Bracelets",
  "Accessories",
];

/* =========================================
   IMAGE URL
========================================= */

function getImageUrl(image) {
  if (!image) return "";

  const value = String(image).trim();

  if (!value) return "";

  if (
    value.startsWith("http://") ||
    value.startsWith("https://") ||
    value.startsWith("data:")
  ) {
    return value;
  }

  if (value.startsWith("/")) {
    return `${API}${value}`;
  }

  return `${API}/${value}`;
}

/* =========================================
   SAFE NUMBER
========================================= */

function safeNumber(value, fallback = 0) {
  const number = Number(value);

  return Number.isFinite(number)
    ? number
    : fallback;
}

/* =========================================
   NORMALIZE PRODUCT
========================================= */

function normalizeProduct(product, index) {
  const colors =
    Array.isArray(product?.colors) &&
    product.colors.length > 0
      ? product.colors
      : ["Black"];

  return {
    ...product,

    id:
      product?.id ??
      product?._id ??
      `product-${index}`,

    name:
      product?.name ||
      "Luxury Bag",

    category:
      product?.category ||
      "Bags",

    description:
      product?.description ||
      "Premium quality product from the LUXORA collection.",

    price: safeNumber(
      product?.price,
      0
    ),

    oldPrice: safeNumber(
      product?.oldPrice,
      safeNumber(product?.price, 0)
    ),

    stock: safeNumber(
      product?.stock,
      0
    ),

    rating: safeNumber(
      product?.rating,
      4.8
    ),

    reviews: safeNumber(
      product?.reviews,
      0
    ),

    colors: colors.map(String),

    image: getImageUrl(
      product?.image
    ),
  };
}

/* =========================================
   APP
========================================= */

function App() {
  const [products, setProducts] = useState([]);

  const [loadingProducts, setLoadingProducts] =
    useState(true);

  const [apiError, setApiError] =
    useState("");

  const [selectedCategory, setSelectedCategory] =
    useState("All Bags");

  const [searchOpen, setSearchOpen] =
    useState(false);

  const [searchText, setSearchText] =
    useState("");

  const [wishlist, setWishlist] =
    useState([]);

  const [selectedProduct, setSelectedProduct] =
    useState(null);

  const [selectedColor, setSelectedColor] =
    useState("Black");

  const [detailQuantity, setDetailQuantity] =
    useState(1);

  const [cart, setCart] =
    useState([]);

  const [cartOpen, setCartOpen] =
    useState(false);

  const [checkoutOpen, setCheckoutOpen] =
    useState(false);

  const [orderPlaced, setOrderPlaced] =
    useState(false);

  const [orderLoading, setOrderLoading] =
    useState(false);

  const [orderReference, setOrderReference] =
    useState("");

  /* =========================================
     PAYMENT
  ========================================= */

  const [paymentMethod, setPaymentMethod] =
    useState("UPI");

  const [selectedBank, setSelectedBank] =
    useState("");

  const [upiId, setUpiId] =
    useState("");

  const [cardDetails, setCardDetails] =
    useState({
      cardNumber: "",
      expiry: "",
      cvv: "",
      name: "",
    });

  /* =========================================
     CUSTOMER
  ========================================= */

  const [customer, setCustomer] =
    useState({
      name: "",
      mobile: "",
      email: "",
      address: "",
      city: "",
      state: "",
      pincode: "",
    });

  /* =========================================
     LOAD PRODUCTS
  ========================================= */

  useEffect(() => {
    let cancelled = false;

    async function loadProducts() {
      try {
        setLoadingProducts(true);
        setApiError("");

        const response =
          await fetch(
            `${API}/api/products`
          );

        if (!response.ok) {
          throw new Error(
            `Products API returned ${response.status}`
          );
        }

        const data =
          await response.json();

        console.log(
          "PRODUCT API:",
          data
        );

        if (
          !data ||
          !data.success ||
          !Array.isArray(
            data.products
          )
        ) {
          throw new Error(
            "Invalid products response."
          );
        }

        const normalized =
          data.products.map(
            normalizeProduct
          );

        if (!cancelled) {
          setProducts(
            normalized
          );
        }
      } catch (error) {
        console.error(
          "PRODUCT LOAD ERROR:",
          error
        );

        if (!cancelled) {
          setProducts([]);

          setApiError(
            "Backend se products load nahi ho rahe."
          );
        }
      } finally {
        if (!cancelled) {
          setLoadingProducts(false);
        }
      }
    }

    loadProducts();

    return () => {
      cancelled = true;
    };
  }, []);

  /* =========================================
     FILTER
  ========================================= */

  const filteredProducts =
    useMemo(() => {
      const search =
        searchText
          .trim()
          .toLowerCase();

      return products.filter(
        (product) => {
          const category =
            String(
              product.category || ""
            ).toLowerCase();

          const selected =
            selectedCategory.toLowerCase();

          const matchesCategory =
            selectedCategory ===
              "All Bags" ||
            selectedCategory ===
              "Bags" ||
            category === selected;

          const searchableText =
            [
              product.name,
              product.category,
              product.description,
            ]
              .filter(Boolean)
              .join(" ")
              .toLowerCase();

          const matchesSearch =
            !search ||
            searchableText.includes(
              search
            );

          return (
            matchesCategory &&
            matchesSearch
          );
        }
      );
    }, [
      products,
      selectedCategory,
      searchText,
    ]);

  /* =========================================
     CART TOTALS
  ========================================= */

  const totalItems =
    cart.reduce(
      (total, item) =>
        total +
        safeNumber(
          item.quantity,
          0
        ),
      0
    );

  const totalPrice =
    cart.reduce(
      (total, item) =>
        total +
        safeNumber(
          item.price,
          0
        ) *
          safeNumber(
            item.quantity,
            0
          ),
      0
    );

  const deliveryCharge =
    totalPrice >= 1999 ||
    totalPrice === 0
      ? 0
      : 99;

  const checkoutTotal =
    totalPrice +
    deliveryCharge;

  /* =========================================
     WISHLIST
  ========================================= */

  function toggleWishlist(id) {
    setWishlist(
      (current) =>
        current.includes(id)
          ? current.filter(
              (itemId) =>
                itemId !== id
            )
          : [
              ...current,
              id,
            ]
    );
  }

  /* =========================================
     PRODUCT
  ========================================= */

  function openProduct(product) {
    setSelectedProduct(
      product
    );

    setSelectedColor(
      product.colors?.[0] ||
        "Black"
    );

    setDetailQuantity(1);
  }

  function closeProduct() {
    setSelectedProduct(null);
  }

  /* =========================================
     ADD TO CART
  ========================================= */

  function addToCart(
    product,
    quantity = 1,
    color = ""
  ) {
    if (!product) return;

    const availableStock =
      safeNumber(
        product.stock,
        0
      );

    if (availableStock <= 0) {
      alert(
        "This product is currently out of stock."
      );
      return;
    }

    const finalColor =
      color ||
      product.colors?.[0] ||
      "Black";

    const finalQuantity =
      Math.max(
        1,
        safeNumber(
          quantity,
          1
        )
      );

    setCart(
      (currentCart) => {
        const existing =
          currentCart.find(
            (item) =>
              String(item.id) ===
                String(
                  product.id
                ) &&
              item.color ===
                finalColor
          );

        if (existing) {
          const newQuantity =
            safeNumber(
              existing.quantity,
              0
            ) +
            finalQuantity;

          if (
            newQuantity >
            availableStock
          ) {
            alert(
              `Only ${availableStock} item(s) available in stock.`
            );

            return currentCart;
          }

          return currentCart.map(
            (item) =>
              String(
                item.id
              ) ===
                String(
                  product.id
                ) &&
              item.color ===
                finalColor
                ? {
                    ...item,
                    quantity:
                      newQuantity,
                  }
                : item
          );
        }

        const safeQuantity =
          Math.min(
            finalQuantity,
            availableStock
          );

        return [
          ...currentCart,
          {
            ...product,
            id: String(
              product.id
            ),
            quantity:
              safeQuantity,
            color:
              finalColor,
          },
        ];
      }
    );

    setCartOpen(true);
  }

  /* =========================================
     CART QUANTITY
  ========================================= */

  function increaseQuantity(
    id,
    color
  ) {
    const product =
      products.find(
        (item) =>
          String(item.id) ===
          String(id)
      );

    const stock =
      safeNumber(
        product?.stock,
        0
      );

    setCart(
      (currentCart) =>
        currentCart.map(
          (item) => {
            if (
              String(
                item.id
              ) !==
                String(id) ||
              item.color !==
                color
            ) {
              return item;
            }

            const newQuantity =
              safeNumber(
                item.quantity,
                0
              ) + 1;

            if (
              stock > 0 &&
              newQuantity >
                stock
            ) {
              alert(
                `Only ${stock} item(s) available in stock.`
              );

              return item;
            }

            return {
              ...item,
              quantity:
                newQuantity,
            };
          }
        )
    );
  }

  function decreaseQuantity(
    id,
    color
  ) {
    setCart(
      (currentCart) =>
        currentCart
          .map(
            (item) =>
              String(
                item.id
              ) ===
                String(id) &&
              item.color ===
                color
                ? {
                    ...item,
                    quantity:
                      safeNumber(
                        item.quantity,
                        0
                      ) - 1,
                  }
                : item
          )
          .filter(
            (item) =>
              safeNumber(
                item.quantity,
                0
              ) > 0
          )
    );
  }

  function removeFromCart(
    id,
    color
  ) {
    setCart(
      (currentCart) =>
        currentCart.filter(
          (item) =>
            !(
              String(
                item.id
              ) ===
                String(id) &&
              item.color ===
                color
            )
        )
    );
  }

  /* =========================================
     CUSTOMER INPUT
  ========================================= */

  function updateCustomer(
    field,
    value
  ) {
    let cleanedValue =
      value;

    if (
      field === "name" ||
      field === "city" ||
      field === "state"
    ) {
      cleanedValue =
        value.replace(
          /[^A-Za-zÀ-ÿ\s.'-]/g,
          ""
        );
    }

    if (
      field === "mobile" ||
      field === "pincode"
    ) {
      cleanedValue =
        value
          .replace(
            /\D/g,
            ""
          )
          .slice(
            0,
            field ===
              "mobile"
              ? 10
              : 6
          );
    }

    if (
      field === "email"
    ) {
      cleanedValue =
        value
          .replace(
            /\s/g,
            ""
          )
          .slice(
            0,
            150
          );
    }

    if (
      field === "address"
    ) {
      cleanedValue =
        value.replace(
          /[^A-Za-z0-9À-ÿ\s.,/#()'&-]/g,
          ""
        );
    }

    setCustomer(
      (current) => ({
        ...current,
        [field]:
          cleanedValue,
      })
    );
  }

  /* =========================================
     CARD
  ========================================= */

  function updateCard(
    field,
    value
  ) {
    let cleanedValue =
      value;

    if (
      field ===
      "cardNumber"
    ) {
      cleanedValue =
        value
          .replace(
            /\D/g,
            ""
          )
          .slice(
            0,
            16
          )
          .replace(
            /(.{4})/g,
            "$1 "
          )
          .trim();
    }

    if (
      field ===
      "expiry"
    ) {
      cleanedValue =
        value
          .replace(
            /\D/g,
            ""
          )
          .slice(
            0,
            4
          );

      if (
        cleanedValue.length >
        2
      ) {
        cleanedValue =
          `${cleanedValue.slice(
            0,
            2
          )}/${cleanedValue.slice(
            2
          )}`;
      }
    }

    if (
      field === "cvv"
    ) {
      cleanedValue =
        value
          .replace(
            /\D/g,
            ""
          )
          .slice(
            0,
            3
          );
    }

    if (
      field === "name"
    ) {
      cleanedValue =
        value.replace(
          /[^A-Za-zÀ-ÿ\s.'-]/g,
          ""
        );
    }

    setCardDetails(
      (current) => ({
        ...current,
        [field]:
          cleanedValue,
      })
    );
  }

  /* =========================================
     PAYMENT METHOD
  ========================================= */

  function changePaymentMethod(
    method
  ) {
    setPaymentMethod(
      method
    );

    setSelectedBank("");
    setUpiId("");

    setCardDetails({
      cardNumber: "",
      expiry: "",
      cvv: "",
      name: "",
    });
  }

  /* =========================================
     CHECKOUT
  ========================================= */

  function openCheckout() {
    if (
      cart.length === 0
    ) {
      alert(
        "Your cart is empty."
      );
      return;
    }

    setCartOpen(false);
    setCheckoutOpen(true);
    setOrderPlaced(false);
  }

  /* =========================================
     PAYMENT VALIDATION
  ========================================= */

  function validatePayment() {
    if (
      paymentMethod ===
      "UPI"
    ) {
      if (
        !upiId.trim()
      ) {
        alert(
          "Please enter your UPI ID."
        );
        return false;
      }

      if (
        !/^[\w.-]+@[\w.-]+$/.test(
          upiId.trim()
        )
      ) {
        alert(
          "Please enter a valid UPI ID."
        );
        return false;
      }
    }

    if (
      paymentMethod ===
        "CREDIT_CARD" ||
      paymentMethod ===
        "DEBIT_CARD"
    ) {
      const cleanCard =
        cardDetails.cardNumber.replace(
          /\s/g,
          ""
        );

      if (
        cleanCard.length !==
        16
      ) {
        alert(
          "Please enter a valid 16-digit card number."
        );
        return false;
      }

      if (
        !/^\d{2}\/\d{2}$/.test(
          cardDetails.expiry
        )
      ) {
        alert(
          "Please enter a valid expiry date."
        );
        return false;
      }

      if (
        cardDetails.cvv.length !==
        3
      ) {
        alert(
          "Please enter a valid CVV."
        );
        return false;
      }

      if (
        !cardDetails.name.trim()
      ) {
        alert(
          "Please enter cardholder name."
        );
        return false;
      }
    }

    if (
      paymentMethod ===
      "NET_BANKING"
    ) {
      if (!selectedBank) {
        alert(
          "Please select your bank."
        );
        return false;
      }
    }

    return true;
  }

  /* =========================================
     PLACE ORDER
  ========================================= */

  async function placeOrder(
    event
  ) {
    event.preventDefault();

    if (
      orderLoading
    ) {
      return;
    }

    if (
      cart.length ===
      0
    ) {
      alert(
        "Your cart is empty."
      );
      return;
    }

    if (
      !customer.name.trim() ||
      !customer.mobile.trim() ||
      !customer.address.trim() ||
      !customer.city.trim() ||
      !customer.state.trim() ||
      !customer.pincode.trim()
    ) {
      alert(
        "Please fill all required delivery details."
      );
      return;
    }

    if (
      !/^[A-Za-zÀ-ÿ\s.'-]{2,100}$/.test(
        customer.name.trim()
      )
    ) {
      alert(
        "Please enter a valid name."
      );
      return;
    }

    if (
      !/^[6-9]\d{9}$/.test(
        customer.mobile.trim()
      )
    ) {
      alert(
        "Please enter a valid 10-digit mobile number."
      );
      return;
    }

    if (
      customer.email.trim() &&
      !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(
        customer.email.trim()
      )
    ) {
      alert(
        "Please enter a valid email address."
      );
      return;
    }

    if (
      customer.address.trim()
        .length < 5
    ) {
      alert(
        "Please enter a complete delivery address."
      );
      return;
    }

    if (
      !/^[A-Za-zÀ-ÿ\s.'-]{2,100}$/.test(
        customer.city.trim()
      )
    ) {
      alert(
        "Please enter a valid city name."
      );
      return;
    }

    if (
      !/^[A-Za-zÀ-ÿ\s.'-]{2,100}$/.test(
        customer.state.trim()
      )
    ) {
      alert(
        "Please enter a valid state name."
      );
      return;
    }

    if (
      !/^[1-9][0-9]{5}$/.test(
        customer.pincode.trim()
      )
    ) {
      alert(
        "Please enter a valid 6-digit PIN code."
      );
      return;
    }

    if (
      !validatePayment()
    ) {
      return;
    }

    try {
      setOrderLoading(true);

      const paymentData = {
        method:
          paymentMethod,

        bank:
          paymentMethod ===
          "NET_BANKING"
            ? selectedBank
            : "",

        upiId:
          paymentMethod ===
          "UPI"
            ? upiId.trim()
            : "",

        card:
          paymentMethod ===
            "CREDIT_CARD" ||
          paymentMethod ===
            "DEBIT_CARD"
            ? {
                last4:
                  cardDetails.cardNumber
                    .replace(
                      /\s/g,
                      ""
                    )
                    .slice(
                      -4
                    ),

                expiry:
                  cardDetails.expiry,

                cardholderName:
                  cardDetails.name.trim(),
              }
            : null,
      };

      /* =====================================
         ORDER DATA
      ===================================== */

      const orderData = {
        customer: {
          name:
            customer.name.trim(),

          phone:
            customer.mobile.trim(),

          email:
            customer.email.trim(),

          address:
            customer.address.trim(),

          city:
            customer.city.trim(),

          state:
            customer.state.trim(),

          pincode:
            customer.pincode.trim(),
        },

        items:
          cart.map(
            (item) => ({
              productId:
                String(
                  item.id
                ),

              name:
                item.name,

              category:
                item.category,

              price:
                safeNumber(
                  item.price,
                  0
                ),

              quantity:
                safeNumber(
                  item.quantity,
                  1
                ),

              color:
                item.color ||
                "Black",

              image:
                item.image ||
                "",
            })
          ),

        subtotal:
          totalPrice,

        deliveryCharge:
          deliveryCharge,

        total:
          checkoutTotal,

        totalItems:
          totalItems,

        payment:
          paymentData,

        paymentMethod:
          paymentMethod ===
          "COD"
            ? "Cash on Delivery"
            : paymentMethod,

        paymentStatus:
          "Pending",

        notes: "",
      };

      console.log(
        "ORDER SENT:",
        orderData
      );

      /* =====================================
         SEND ORDER
      ===================================== */

      const response =
        await fetch(
          `${API}/api/orders`,
          {
            method:
              "POST",

            headers: {
              "Content-Type":
                "application/json",
            },

            body:
              JSON.stringify(
                orderData
              ),
          }
        );

      let data =
        null;

      try {
        data =
          await response.json();
      } catch {
        data =
          null;
      }

      console.log(
        "ORDER RESPONSE:",
        data
      );

      if (!response.ok) {
        throw new Error(
          data?.message ||
            `Order API error ${response.status}`
        );
      }

      if (
        data &&
        data.success ===
          false
      ) {
        throw new Error(
          data.message ||
            "Order could not be placed."
        );
      }

      const reference =
        data?.order
          ?.orderNumber ||
        data?.order
          ?.orderId ||
        data?.order?.id ||
        data?.orderNumber ||
        data?.id ||
        `LUX${Date.now()
          .toString()
          .slice(-6)}`;

      setOrderReference(
        reference
      );

      setOrderPlaced(
        true
      );
    } catch (error) {
      console.error(
        "PLACE ORDER ERROR:",
        error
      );

      alert(
        "Order place nahi hua.\n\n" +
          (
            error.message ||
            "Unknown error"
          )
      );
    } finally {
      setOrderLoading(
        false
      );
    }
  }

  /* =========================================
     CATEGORY
  ========================================= */

  function selectCategory(
    category
  ) {
    setSelectedCategory(
      category
    );

    setTimeout(() => {
      document
        .getElementById(
          "products"
        )
        ?.scrollIntoView({
          behavior:
            "smooth",
        });
    }, 100);
  }

  /* =========================================
     IMAGE FALLBACK
  ========================================= */

  function handleImageError(
    event
  ) {
    event.currentTarget.style.display =
      "none";

    const parent =
      event.currentTarget
        .parentElement;

    if (
      parent &&
      !parent.querySelector(
        ".photo-placeholder"
      )
    ) {
      const fallback =
        document.createElement(
          "div"
        );

      fallback.className =
        "photo-placeholder";

      fallback.innerText =
        "LUXORA";

      parent.appendChild(
        fallback
      );
    }
  }

  return (

            Collection

          <button

            onClick={() =>
              setSearchOpen(
                (value) =>
                  !value
              )
            }
          >
            Search
          </button>

          <button
            type="button"
            onClick={() =>
              document
                .getElementById(
                  "products"
                )
                ?.scrollIntoView({
                  behavior:
                    "smooth",
                })
            }
          >
            Wishlist
            {wishlist.length >
            0

        </div>

      </header>

      {/* SEARCH */}

                              </strong>

                              <small>
                                {
                                  option[2]
                                }
                              </small>

                            </span>

                            <i>
                              {paymentMethod ===
                              option[0]
                                ? "✓"
                                : ""}
                            </i>

                          </button>
                        )
                      )}

                    </div>

                    {/* UPI */}

                    {paymentMethod ===
                      "UPI" && (

                      <div className="dynamic-payment-box">

                        <div className="upi-layout">

                          <div className="upi-form">

                            <span className="mini-label">
                              UPI PAYMENT
                            </span>

                            <h4>
                              Pay securely using UPI
                            </h4>

                            <p>
                              Enter your UPI ID and continue.
                            </p>

                            <label>
                              UPI ID
                            </label>

                            <input
                              type="text"
                              value={
                                upiId
                              }
                              onChange={(event) =>
                                setUpiId(
                                  event.target.value
                                )
                              }
                              placeholder="example@upi"
                              autoComplete="off"
                            />

                            <div className="upi-apps">

                              <span>
                                G
                              </span>

                              <span>
                                P
                              </span>

                              <span>
                                ₹
                              </span>

                              <small>
                                Google Pay • PhonePe • Paytm
                              </small>

                            </div>

                          </div>

                          <div className="qr-box">

                            <div className="fake-qr">

                              <span>
                                LUXORA
                              </span>

                              <div className="qr-pattern">
                                ▦ ▦ ▦
                                <br />
                                ▦ ▦ ▦
                                <br />
                                ▦ ▦ ▦
                              </div>

                            </div>

                            <small>
                              Scan to pay
                            </small>

                          </div>

                        </div>

                      </div>
                    )}

                    {/* CARD */}

                    {(paymentMethod ===
                      "CREDIT_CARD" ||
                      paymentMethod ===
                        "DEBIT_CARD") && (

                      <div className="dynamic-payment-box">

                        <div className="card-preview">

                          <div className="card-top">

                            <span>
                              LUXORA
                            </span>

                            <span>
                              {paymentMethod ===
                              "CREDIT_CARD"
                                ? "CREDIT"
                                : "DEBIT"}
                            </span>

                          </div>

                          <div className="card-number-preview">

                            {cardDetails.cardNumber ||
                              "•••• •••• •••• ••••"}

                          </div>

                          <div className="card-bottom">

                            <span>
                              {cardDetails.name ||
                                "CARDHOLDER NAME"}
                            </span>

                            <span>
                              {cardDetails.expiry ||
                                "MM/YY"}
                            </span>

                          </div>

                        </div>

                        <div className="form-grid card-fields">

                          <div className="form-group full">

                            <label>
                              Card Number *
                            </label>

                            <input
                              type="text"
                              value={
                                cardDetails.cardNumber
                              }
                              onChange={(event) =>
                                updateCard(
                                  "cardNumber",
                                  event.target.value
                                )
                              }
                              inputMode="numeric"
                              autoComplete="cc-number"
                              placeholder="1234 5678 9012 3456"
                              required
                            />

                          </div>

                          <div className="form-group">

                            <label>
                              Expiry Date *
                            </label>

                            <input
                              type="text"
                              value={
                                cardDetails.expiry
                              }
                              onChange={(event) =>
                                updateCard(
                                  "expiry",
                                  event.target.value
                                )
                              }
                              inputMode="numeric"
                              placeholder="MM/YY"
                              required
                            />

                          </div>

                          <div className="form-group">

                            <label>
                              CVV *
                            </label>

                            <input
                              type="password"
                              value={
                                cardDetails.cvv
                              }
                              onChange={(event) =>
                                updateCard(
                                  "cvv",
                                  event.target.value
                                )
                              }
                              inputMode="numeric"
                              autoComplete="cc-csc"
                              placeholder="•••"
                              maxLength="3"
                              required
                            />

                          </div>

                          <div className="form-group full">

                            <label>
                              Cardholder Name *
                            </label>

                            <input
                              type="text"
                              value={
                                cardDetails.name
                              }
                              onChange={(event) =>
                                updateCard(
                                  "name",
                                  event.target.value
                                )
                              }
                              autoComplete="cc-name"
                              placeholder="Name as shown on card"
                              required
                            />

                          </div>

                        </div>

                      </div>
                    )}

                    {/* NET BANKING */}

                    {paymentMethod ===
                      "NET_BANKING" && (

                      <div className="dynamic-payment-box">

                        <span className="mini-label">
                          NET BANKING
                        </span>

                        <h4>
                          Select your bank
                        </h4>

                        <div className="bank-grid">

                          {[
                            "HDFC Bank",
                            "ICICI Bank",
                            "State Bank of India",
                            "Axis Bank",
                            "Kotak Mahindra Bank",
                            "Bank of Baroda",
                            "Punjab National Bank",
                            "Other Banks",
                          ].map(
                            (bank) => (
                              <button
                                type="button"
                                key={bank}
                                className={
                                  selectedBank ===
                                  bank
                                    ? "bank-option active"
                                    : "bank-option"
                                }
                                onClick={() =>
                                  setSelectedBank(
                                    bank
                                  )
                                }
                              >

                                <span className="bank-logo">
                                  {bank
                                    .split(
                                      " "
                                    )
                                    .map(
                                      (word) =>
                                        word[0]
                                    )
                                    .slice(
                                      0,
                                      2
                                    )
                                    .join("")}
                                </span>

                                <span>
                                  {
                                    bank
                                  }
                                </span>

                                {selectedBank ===
                                  bank && (
                                  <b>
                                    ✓
                                  </b>
                                )}

                              </button>
                            )
                          )}

                        </div>

                      </div>
                    )}

                    {/* COD */}

                    {paymentMethod ===
                      "COD" && (

                      <div className="dynamic-payment-box cod-box">

                        <div className="cod-icon-large">
                          ₹
                        </div>

                        <div>

                          <span className="mini-label">
                            CASH ON DELIVERY
                          </span>

                          <h4>
                            Pay when your order arrives
                          </h4>

                          <p>
                            Pay the delivery partner
                            when your LUXORA order
                            reaches your doorstep.
                          </p>

                          <div className="cod-points">

                            <span>
                              ✓ No online payment required
                            </span>

                            <span>
                              ✓ Order confirmation instantly
                            </span>

                          </div>

                        </div>

                      </div>
                    )}

                    {/* SECURITY */}

                    <div className="payment-trust">

                      <div>
                        🔒
                        <span>
                          <strong>
                            Secure Checkout
                          </strong>

                          <small>
                            Your information is encrypted
                          </small>
                        </span>
                      </div>

                      <div>
                        ✓
                        <span>
                          <strong>
                            Trusted Payment
                          </strong>

                          <small>
                            Safe & reliable payment options
                          </small>
                        </span>
                      </div>

                      <div>
                        ↩
                        <span>
                          <strong>
                            Easy Returns
                          </strong>

                          <small>
                            Hassle-free return experience
                          </small>
                        </span>
                      </div>

                    </div>

                  </div>

                  {/* MOBILE BUTTON */}

                  <button
                    type="submit"
                    className="place-order-button mobile-place-order"
                    disabled={
                      orderLoading
                    }
                  >
                    {orderLoading
                      ? "PROCESSING..."
                      : `PLACE ORDER • ₹${checkoutTotal.toLocaleString(
                          "en-IN"
                        )}`}
                  </button>

                </form>

                {/* RIGHT SUMMARY */}

                <aside className="order-summary">

                  <div className="summary-header">

                    <div>

                      <p className="checkout-label">
                        YOUR BAG
                      </p>

                      <h3>
                        Order Summary
                      </h3>

                    </div>

                    <span className="summary-count">
                      {totalItems} Items
                    </span>

                  </div>

                  <div className="summary-products">

                    {cart.map(
                      (item) => (

                        <div
                          className="summary-item"
                          key={`${item.id}-${item.color}`}
                        >

                          <div className="summary-product-image">

                            {item.image ? (
                              <img
                                src={
                                  item.image
                                }
                                alt={
                                  item.name
                                }
                                onError={
                                  handleImageError
                                }
                              />
                            ) : (
                              <span>
                                LUXORA
                              </span>
                            )}

                            <b>
                              {
                                item.quantity
                              }
                            </b>

                          </div>

                          <div className="summary-product-info">

                            <strong>
                              {
                                item.name
                              }
                            </strong>

                            <span>
                              {
                                item.category
                              }
                            </span>

                            <span>
                              Color:{" "}
                              {
                                item.color
                              }
                            </span>

                          </div>

                          <b className="summary-product-price">
                            ₹
                            {(
                              safeNumber(
                                item.price,
                                0
                              ) *
                              safeNumber(
                                item.quantity,
                                0
                              )
                            ).toLocaleString(
                              "en-IN"
                            )}
                          </b>

                        </div>
                      )
                    )}

                  </div>

                  <div className="summary-offer">

                    <span>
                      ✦
                    </span>

                    <div>

                      <strong>
                        Free Delivery
                      </strong>

                      <small>
                        Orders above ₹1,999
                      </small>

                    </div>

                  </div>

                  <div className="summary-calculation">

                    <div>

                      <span>
                        Subtotal
                      </span>

                      <strong>
                        ₹
                        {totalPrice.toLocaleString(
                          "en-IN"
                        )}
                      </strong>

                    </div>

                    <div>

                      <span>
                        Delivery
                      </span>

                      <strong>
                        {deliveryCharge ===
                        0
                          ? "FREE"
                          : `₹${deliveryCharge}`}
                      </strong>

                    </div>

                  </div>

                  <div className="summary-total">

                    <span>
                      Total Amount
                    </span>

                    <strong>
                      ₹
                      {checkoutTotal.toLocaleString(
                        "en-IN"
                      )}
                    </strong>

                  </div>

                  <button
                    type="submit"
                    form="luxora-checkout-form"
                    className="place-order-button desktop-place-order"
                    disabled={
                      orderLoading
                    }
                  >
                    {orderLoading
                      ? "PROCESSING..."
                      : "PLACE ORDER"}
                  </button>

                  <div className="secure-summary">

                    <span>
                      🔒
                    </span>

                    <div>

                      <strong>
                        100% Secure Payment
                      </strong>

                      <small>
                        Your payment details are protected.
                      </small>

                    </div>

                  </div>

                  <div className="accepted-payments">

                    <small>
                      WE ACCEPT
                    </small>

                    <div>

                      <span>
                        UPI
                      </span>

                      <span>
                        VISA
                      </span>

                      <span>
                        MC
                      </span>

                      <span>
                        RuPay
                      </span>

                    </div>

                  </div>

                </aside>

              </div>
            )}

          </div>

        </div>
      )}

    </div>
  );
}

export default App;
